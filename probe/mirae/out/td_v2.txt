// Função para adicionar badge "NOVO" e classe no pai
function adicionarBadgeNovo(urlParcial) {
    // Aguarda o carregamento do DOM
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', () => adicionarBadgeNovo(urlParcial));
        return;
    }

    // Cria o elemento span com o badge "NOVO"
    function criarBadgeNovo() {
        const badge = document.createElement('span');
        badge.className = 'td-menu-badge td-menu-badge--novo';
        badge.textContent = 'NOVO';
        return badge;
    }

    // Processa um link individual
    function processarLink(link) {
        // Verifica se o href contém a URL parcial
        if (link.href && link.href.includes(urlParcial)) {
            const itemPai = link.closest('li') || link.parentElement;
            
            // Adiciona classe no elemento pai
            if (itemPai && !itemPai.classList.contains('td-menu-item--has-badge')) {
                itemPai.classList.add('td-menu-item--has-badge');
                itemPai.classList.add('td-menu-item--novo');
            }
            
            // Verifica se já não existe um badge
            if (!link.parentElement.querySelector('.td-menu-badge--novo')) {
                const badge = criarBadgeNovo();
                
                // Insere o badge como irmão do link
                link.append(badge);
            }
        }
    }

    // Busca e processa links no menu desktop
    function processarMenuDesktop() {
        const linksDesktop = document.querySelectorAll('#desktopNav a, .td-header__nav a');
        linksDesktop.forEach(processarLink);
    }

    // Busca e processa links no menu mobile
    function processarMenuMobile() {
        const linksMobile = document.querySelectorAll('#mobileNav a, .td-mobile-menu a');
        linksMobile.forEach(processarLink);
    }

    // Processa ambos os menus
    processarMenuDesktop();
    processarMenuMobile();

    // Observer para detectar mudanças dinâmicas no menu
    const observer = new MutationObserver(() => {
        processarMenuDesktop();
        processarMenuMobile();
    });

    // Observa mudanças no nav desktop e mobile
    const navDesktop = document.getElementById('desktopNav');
    const navMobile = document.getElementById('mobileNav');

    if (navDesktop) {
        observer.observe(navDesktop, { childList: true, subtree: true });
    }

    if (navMobile) {
        observer.observe(navMobile, { childList: true, subtree: true });
    }
}
